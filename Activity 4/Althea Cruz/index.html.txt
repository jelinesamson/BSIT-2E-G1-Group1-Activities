<script>
const mobileInput = document.getElementById("mobile");

mobileInput.addEventListener("input", function () {
    this.value = this.value.replace(/[^0-9]/g, "");
});

document.getElementById("dataForm").addEventListener("submit", function (e) {
    e.preventDefault();

    const warning = document.getElementById("warningMsg");

    const fullname = document.getElementById("fullname").value.trim();
    if (fullname === "") {
        warning.textContent = "Please enter your fullname";
        warning.style.display = "block";
        return;
    }

    const dob = document.getElementById("dob").value.trim();
    if (dob === "") {
        warning.textContent = "Please enter your date of birth";
        warning.style.display = "block";
  





return;
    }

    const gender = document.getElementById("gender").value.trim();
    if (gender === "") {
        warning.textContent = "Please select your gender";
        warning.style.display = "block";
        return;
    }

    const status = document.getElementById("status").value.trim();
    if (status === "") {
        warning.textContent = "Please select your civil status";
        warning.style.display = "block";
        return;
    }

    const email = document.getElementById("email").value.trim();
    if (email === "") {
        warning.textContent = "Please enter your email address";
        warning.style.display = "block";
        return;
    }

    const mobile = mobileInput.value.trim();
    if (mobile === "") {
        warning.textContent = "Please enter your mobile number";
        warning.style.display = "block";
        return;
    }

    const address = document.getElementById("address").value.trim();
    if (address === "") {
        warning.textContent = "Please enter your complete address";
        warning.style.display = "block";
        return;
    }

    warning.style.display = "none";

    document.getElementById("output").innerHTML = `
        <p><strong>Full Name:</strong> ${fullname}</p>
        <p><strong>Date of Birth:</strong> ${dob}</p>
        <p><strong>Gender:</strong> ${gender}</p>
        <p><strong>Status:</strong> ${status}</p>
        <p><strong>Email:</strong> ${email}</p>
        <p><strong>Mobile:</strong> ${mobile}</p>
        <p><strong>Address:</strong> ${address}</p>
    `;

    document.getElementById("outputCard").style.display = "block";
});
</script>
